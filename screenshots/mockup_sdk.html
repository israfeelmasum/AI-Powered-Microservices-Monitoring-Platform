SDK Integration Guide
Connect any FastAPI microservice in under 5 minutes · No configuration required

Drop-in Middleware
One line of code to track all requests, latency, and error rates automatically across any FastAPI service.

Copy the SDK into your service

2

Run pip install aiohttp python-dotenv — no other packages needed.

Add middleware and start your service

Integration Code
Python · FastAPI
main.py — Add to your FastAPI service
from contextlib import asynccontextmanager
import os
from fastapi import FastAPI
from monitoring_sdk.middleware import MonitoringMiddleware
from monitoring_sdk.client import MonitoringClient

# Lifespan: register on startup, cleanup on shutdown
@asynccontextmanager
async def lifespan(app: FastAPI):
    client = MonitoringClient(
        monitoring_url=os.getenv("MONITORING_URL"),
        service_name="my-service",
        service_secret=os.getenv("MY_SERVICE_SECRET")
    )
    await client.register(
        service_url=f"http://localhost:8005",
        version="1.0.0",
        endpoints=["/health", "/api/items"]
    )
    yield

app = FastAPI(lifespan=lifespan)

# One line — tracks all requests automatically
app.add_middleware(
    MonitoringMiddleware,
    monitoring_url=os.getenv("MONITORING_URL"),
    service_name="my-service",
    service_secret=os.getenv("MY_SERVICE_SECRET")
)
.env — Environment variables
# Monitoring platform URL
MONITORING_URL=http://localhost:8000

# Generate with: python -c "import secrets; print(secrets.token_urlsafe(32))"
MY_SERVICE_SECRET=your-generated-secret-here
